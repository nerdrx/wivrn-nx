cmake_minimum_required(VERSION 3.22)
project(motion_gpu_truth LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_program(GLSLC glslc REQUIRED)
get_filename_component(SRC_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(BENCH_ROOT "" CACHE PATH "NX Warp bench directory with build-host/libnxbench_core.a")
if(NOT EXISTS "${BENCH_ROOT}/build-host/libnxbench_core.a")
  message(FATAL_ERROR "Set BENCH_ROOT to a built NX Warp bench directory")
endif()
set(GEN ${CMAKE_CURRENT_BINARY_DIR}/spv)
file(MAKE_DIRECTORY ${GEN})
foreach(pair IN ITEMS downsample:motion_downsample estimate:motion_estimate warp:motion_warp)
  string(REPLACE : ";" p "${pair}")
  list(GET p 0 n)
  list(GET p 1 s)
  add_custom_command(OUTPUT ${GEN}/${n}.spv
    COMMAND ${GLSLC} -O -I${SRC_ROOT}/server/compositor/shaders -o ${GEN}/${n}.spv ${SRC_ROOT}/server/compositor/shaders/${s}.comp
    DEPENDS ${SRC_ROOT}/server/compositor/shaders/${s}.comp ${SRC_ROOT}/server/compositor/shaders/motion_constants.glsl.inc)
  list(APPEND SPV ${GEN}/${n}.spv)
endforeach()
add_custom_target(shaders DEPENDS ${SPV})
add_executable(motion_gpu_truth main.cpp)
add_dependencies(motion_gpu_truth shaders)
target_include_directories(motion_gpu_truth PRIVATE ${BENCH_ROOT}/src/core)
target_include_directories(motion_gpu_truth PRIVATE /usr/include ${BENCH_ROOT}/../../tools/Vulkan-Headers-1.4.309/include)
target_link_libraries(motion_gpu_truth PRIVATE ${BENCH_ROOT}/build-host/libnxbench_core.a vulkan)
target_compile_definitions(motion_gpu_truth PRIVATE SPV_DIR=\"${GEN}\")

set(MOTION_TRUTH_SIZE 256 CACHE STRING "Square fixture size, multiple of 64")
target_compile_definitions(motion_gpu_truth PRIVATE MOTION_TRUTH_SIZE=${MOTION_TRUTH_SIZE})

set(MOTION_TRUTH_BLOCK 64 CACHE STRING "Motion cell size in pixels")
target_compile_definitions(motion_gpu_truth PRIVATE MOTION_TRUTH_BLOCK=${MOTION_TRUTH_BLOCK})
